kigit Diff

How to use

Arguments

There are numerous ways to use kigit diff with different arguments:

  • In order to check uncommited changes to the last commit, use: kigit diff
  • In order to check uncommited changes of a certain file to the last commit, use: kigit diff <filename>
  • In order to check the latest commit to an earlier commit, use: kigit diff <commit>
  • In order to check the changes of only a certain file within the latest commit to the file within an earlier commit, use: kigit diff <commit> <filename>
  • In order to check the changes between two specific commits, use: kigit diff <commit> <commit>
  • You can also check changes between branches, tags, or a combination of tags, branches and commits, by using kigit diff <commit/branch/tag> <commit/branch/tag>, or simply kigit diff <commit/branch/tag>

Options

In addition to arguments, kigit diff also has 3 different options to work with:

  • The option --show-filename: boolean lets kigit display the changed components of each file in their own respective row, and adds the filename of each file to the depicted table.
  • The option --detail: str causes kigit to show a detailed view of the modified components of Kicad files. It is a string value, allowing the user to type in what detail is supposed to be shown.
  • The option --debug: boolean enables additional debug options for the kigit output.

@app.command()
def diff(
        args: Annotated[Optional[List[str]], typer.Argument(help="commit file or file or commit")] = None,
        show_filename: bool = typer.Option(False, help="Shows corresponding Filename in result"),
        detail: str = typer.Option("", "--detail", help="Filters detail screen"),
        debug: bool = typer.Option(False, help="enables Debug options")
        ):
    """Show changes between commits, commit and working tree and more for KiCad files"""

    #default values
    commit = ["HEAD", None]
    file = None

    if args:
        if len(args) == 1:
            val = args[0]
            if re.search(r"\.kicad_.*$", val) or "/" in val:
                file = val
            else:
                commit[0] = val
                file = None
        elif len(args) == 2:
            val = args[1]
            if re.search(r"\.kicad_.*$", val) or "/" in val:
                commit[0] = args[0]
                file = val
            else:
                commit[0] = args[0]
                commit[1] = args[1]
        elif len(args) == 3:
            commit[0] = args[0]
            commit[1] = args[1]
            file = args[2]

    #object creation
    gitmanager = GitManager(cwd)

    parser = KiCadDiffParser()

    diff_result = gitmanager.diff(commit[0], commit[1], file)

    if isinstance(diff_result, str):
        print(f"[ERROR] Git: {diff_result}")
        raise typer.Exit(1)
    else:
        stats = parser.parse(diff_result, working_tree_dir=gitmanager.repo.working_tree_dir)

    if stats.isEmpty():
        typer.echo("No KiCad file changes detected")
        raise typer.Exit(1)

    if debug:
        print(f"[DEBUG] {stats}")

    if not detail:
        table = Table(title="kigit diff")
        table.add_column("Part", justify="center", style="white")
        table.add_column("Added", justify="center", style="green")
        table.add_column("Removed", justify="center", style="red")
        table.add_column("Modified", justify="center", style="yellow")
        if show_filename:
            table.add_column("Filename", justify="center", style="white")

        if not show_filename:
            for components in stats.groupByComponent():
                table.add_row(str(components.changes[0].reference),
                              str(len(components.getAdded())),
                              str(len(components.getRemoved())),
                              str(len(components.getModified())))
        else:
            for components in stats.filterFile():
                for entry in components.groupByComponent():
                    table.add_row(str(entry.changes[0].reference),
                                  str(len(entry.getAdded())),
                                  str(len(entry.getRemoved())),
                                  str(len(entry.getModified())),
                                  str(entry.changes[0].origin))


        console.print(table)
    else:
        table = Table(title="KiGit Diff Detail Screen")
        table.add_column("Part", justify="center")
        match detail.lower():
            case "modified":
                table.add_column("Old Position", justify="center")
                table.add_column("New Position", justify="center")
                modified = stats.getModified()
                for entry in modified:
                    if isinstance(entry, bc.ComponentMoved):
                        table.add_row(str(entry.reference),
                                      str(entry.coordinates[0]),
                                      str(entry.coordinates[1]))
        console.print(table)
